SCORE FORMULA — implement as calcScore(tags) function:

function calcScore(tags) {
  const categories = {
    wifi:        { weight: 0.25, max: 10, pts: 0 },
    outlets:     { weight: 0.20, max: 10, pts: 0 },
    environment: { weight: 0.20, max: 10, pts: 0 },
    hours:       { weight: 0.15, max: 10, pts: 0 },
    food:        { weight: 0.10, max: 10, pts: 0 },
    access:      { weight: 0.10, max: 10, pts: 0 }
  };

  const tagMap = {
    wifi_fast:            ['wifi', 4],
    wifi_reliable:        ['wifi', 3],
    wifi_password_free:   ['wifi', 2],
    cell_signal_strong:   ['wifi', 1],
    outlets_every_seat:   ['outlets', 5],
    outlets_ample:        ['outlets', 3],
    outlets_limited:      ['outlets', 1],
    standing_desk:        ['outlets', 1],
    noise_quiet:          ['environment', 4],
    noise_moderate:       ['environment', 2],
    natural_light:        ['environment', 2],
    temp_controlled:      ['environment', 1],
    seating_comfortable:  ['environment', 1],
    table_space:          ['environment', 2],
    open_early:           ['hours', 2],
    open_late:            ['hours', 2],
    no_time_limit:        ['hours', 3],
    walk_in:              ['hours', 2],
    reservation_available:['hours', 1],
    coffee_quality:       ['food', 3],
    food_available:       ['food', 3],
    water_refill:         ['food', 2],
    alcohol_available:    ['food', 2],
    ada_entrance:         ['access', 3],
    ada_restroom:         ['access', 2],
    transit_nearby:       ['access', 2],
    bike_rack:            ['access', 2],
    parking_free:         ['access', 1]
  };

  tags.forEach(tag => {
    if (tagMap[tag]) {
      const [cat, pts] = tagMap[tag];
      categories[cat].pts = Math.min(categories[cat].max, categories[cat].pts + pts);
    }
  });

  const raw = Object.values(categories).reduce((sum, c) => {
    return sum + (c.pts / c.max) * c.weight;
  }, 0);

  return Math.round(raw * 50) / 10; // returns 0.0–5.0
}

SCORE DISPLAY — implement these two helper functions:

function scoreToDots(score) {
  const filled = Math.round(score);
  return '●'.repeat(filled) + '○'.repeat(5 - filled);
}

function scoreToLabel(score) {
  if (score >= 4.5) return 'Excellent';
  if (score >= 3.5) return 'Good';
  if (score >= 2.5) return 'Decent';
  return 'Limited';
}
